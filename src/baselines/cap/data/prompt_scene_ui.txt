objects = ['tomato', 'potato', 'pan', 'stoveburner', 'stoveknob', 'faucet', 'fridge', 'apple', 'sinkbasin', 'drawer', 'book', 'coffeemachine', 'mug', 'egg', 'bread', 'microwave', 'plate', 'countertop', 'kettle',]
# make me a Toast and set the table for lunch
# make Toast
say('OK - toast the bread')
bread_id = get_obj_id('bread')
# It's good practice to always check if an object was found.
if bread_id:
    toaster_id = get_obj_id('toaster')
    move_to(bread_id)
    if parse_question('is bread sliceable?'):
      if not parse_question('is bread sliced?'):
        slice(bread_id)
        breadsliced_id = get_obj_id('breadsliced')
      else:
        breadsliced_id = get_obj_id('breadsliced')
      pickup(breadsliced_id)
      microwave_id = get_obj_id('microwave')
      if microwave_id:
          move_to(microwave_id)
          if parse_question(f'is microwave close?'):
            open(microwave_id)
          put(microwave_id)
          close(microwave_id) 
          toggle_on(microwave_id)
          time.sleep(1)
          toggle_off(microwave_id)
          open(microwave_id)
          pickup(breadsliced_id)
          countertop_id = get_obj_id('countertop')
          move_to(countertop_id)
          put(countertop_id)
      else:
          say("I can't find the microwave.")
    else:
      say('The Bread is not sliceable.')
else:
    say("I can't find the bread.")
# set the table for lunch
plate_id = get_obj_id('plate')
if plate_id:
    move_to(plate_id)
    pickup(plate_id)
    countertop_id = get_obj_id('countertop')
    move_to(countertop_id)
    put(countertop_id)
    tomato_id = get_obj_id('tomato')
    move_to(tomato_id)
    pickup(tomato_id)
    move_to(plate_id)
    put(plate_id)
    breadsliced_id = get_obj_id('breadsliced')
    if breadsliced_id:
        move_to(breadsliced_id)
        pickup(breadsliced_id)
        move_to(plate_id)
        put(plate_id)

    say("I can't find the plate.")
